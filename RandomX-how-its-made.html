One iteration · spec 4.6.2 verbatim

// ic starts at 2048; r0…r7 start at 0
1. spAddr0 ⊕= lo32(r[s0] ⊕ r[s1])
   spAddr1 ⊕= hi32(r[s0] ⊕ r[s1])
2. r0…r7 ⊕= P[spAddr0 & L3_64] // 8 × u64
3. f0…e3 ← P[spAddr1 & L3_64] // IEEE pairs
4. execute ProgramBuffer[0..255]
5. mt ← ma; mx ⊕= lo32(r[s2] ⊕ r[s3])
6. prefetch D[offset + mx % |D|]
7. r0…r7 ⊕= D[offset + mt % |D|] // 64 B DRAM
8. swap(mx, ma)
9. P[spAddr1] ← r0…r7
10. fi ← fi ⊕ ei // v1
11. P[spAddr0] ← f0…f3
12. spAddr0, spAddr1 ← 0
13. ic ← ic − 1
